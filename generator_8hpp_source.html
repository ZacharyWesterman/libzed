490 template <typename U = T, typename std::enable_if<is_iterator<U>::value, int>::type = 0>
491 auto flatten() noexcept {
492 struct unchunkData {
493 std::optional<T> data;
494 std::optional<iterator_value<T>> iter;
495 generator<T, S> gen;
496 };
497
498 auto unchunkFn = [](unchunkData &state) -> std::optional<dereference<T>> {
499 // If the generator has not been initialized,
500 // or there is no more data in the chunk,
501 // then try to generate another chunk.
502 if (!state.data || (state.iter.value() == state.data.value().end())) {
503 // If there's nothing left, we're done.
504 auto chunkData = state.gen.next();
505 if (!chunkData) {
506 return {};
507 }
508
509 auto &val = chunkData.value();
510 // Make sure we didn't just get an empty iterator.
511 if (val.begin() == val.end()) {
512 return {};
513 }
514
515 state.data = chunkData;
516 state.iter = state.data.value().begin();
517 }
518
519 // At this point the generator has been initialized, and current chunk has data in it,
520 // so spit out the next value on the chunk.
521
522 return *state.iter.value()++;
523 };
524
525 return generator<dereference<T>, unchunkData>({{}, {}, *this}, unchunkFn);
526 }
